cmake_minimum_required(VERSION 3.31)
project(JsonInstalledConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiSDKCore TRUE)
find_package(iiJson 0.1 CONFIG REQUIRED COMPONENTS Codec Schema Ontology)
if(TARGET Qt6::Core OR TARGET iiSDKCore::iiSDKCore OR TARGET iiSDKCore::Runtime OR TARGET iiSDKCore::Contracts OR TARGET Threads::Threads OR TARGET iiFileProvider::Native)
    message(FATAL_ERROR "JSON codec imported runtime, storage or GUI dependencies")
endif()
get_target_property(_json_dependencies iiJson::iiJson INTERFACE_LINK_LIBRARIES)
if("iiJson::Ontology" IN_LIST _json_dependencies OR "iiJsonOntology" IN_LIST _json_dependencies)
    message(FATAL_ERROR "Base JSON codec depends on the ontology layer")
endif()
add_executable(consumer ../contracts.cpp)
target_link_libraries(consumer PRIVATE iiJson::iiJson)
enable_testing()
add_test(NAME iiJson.InstalledConsumer COMMAND consumer)
add_executable(duplicate_policy_consumer ../duplicate_policy.cpp)
target_link_libraries(duplicate_policy_consumer PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledDuplicatePolicy COMMAND duplicate_policy_consumer)
add_executable(legacy_duplicate_example ../../example/legacy-duplicate-policy.cpp)
target_link_libraries(legacy_duplicate_example PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledLegacyDuplicateExample COMMAND legacy_duplicate_example)
add_executable(independent_value ../independent_value.cpp)
target_link_libraries(independent_value PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledIndependentValue COMMAND independent_value)
add_executable(schema_consumer ../schema.cpp)
target_link_libraries(schema_consumer PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledSchema COMMAND schema_consumer)
add_executable(domain_example ../../example/domain-schema.cpp)
target_link_libraries(domain_example PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledDomainExample COMMAND domain_example)
add_executable(schema_definition ../schema-definition.cpp)
target_link_libraries(schema_definition PRIVATE iiJson::iiJson)
add_test(NAME iiJson.InstalledSchemaDefinition COMMAND schema_definition)
add_executable(ontology_consumer ../ontology.cpp)
target_link_libraries(ontology_consumer PRIVATE iiJson::Ontology)
add_test(NAME iiJson.InstalledOntology COMMAND ontology_consumer)
add_executable(ontology_example ../../example/creation-ontology.cpp)
target_link_libraries(ontology_example PRIVATE iiJson::Ontology)
add_test(NAME iiJson.InstalledOntologyExample COMMAND ontology_example)
