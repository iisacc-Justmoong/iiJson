cmake_minimum_required(VERSION 3.31)
project(iiJson VERSION 0.1.2 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_library(iiJson STATIC src/iiJson/Json.cpp src/iiJson/Value.cpp src/iiJson/Schema.cpp src/iiJson/SchemaDefinition.cpp)
add_library(iiJson::iiJson ALIAS iiJson)
add_library(iiJsonOntology STATIC src/iiJson/Ontology.cpp src/iiJson/OntologyCodec.cpp src/iiJson/Snapshot.cpp)
add_library(iiJson::Ontology ALIAS iiJsonOntology)
set_target_properties(iiJsonOntology PROPERTIES EXPORT_NAME Ontology CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_link_libraries(iiJsonOntology PUBLIC iiJson::iiJson)
target_compile_features(iiJson PUBLIC cxx_std_23)
target_include_directories(iiJson PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(iiJson PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
add_custom_target(iiJsonBoundary ALL COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
add_dependencies(iiJson iiJsonBoundary)
if(MSVC)
    target_compile_options(iiJson PUBLIC /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(iiJson_contracts tests/contracts.cpp)
    target_link_libraries(iiJson_contracts PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.Contracts COMMAND iiJson_contracts)
    add_executable(iiJson_duplicate_policy tests/duplicate_policy.cpp)
    target_link_libraries(iiJson_duplicate_policy PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.DuplicatePolicy COMMAND iiJson_duplicate_policy)
    add_executable(iiJson_independent_value tests/independent_value.cpp)
    target_link_libraries(iiJson_independent_value PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.IndependentValue COMMAND iiJson_independent_value)
    add_executable(iiJson_schema_contracts tests/schema.cpp)
    target_link_libraries(iiJson_schema_contracts PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.Schema COMMAND iiJson_schema_contracts)
    add_executable(iiJson_schema_definition tests/schema-definition.cpp)
    target_link_libraries(iiJson_schema_definition PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.SchemaDefinition COMMAND iiJson_schema_definition)
    add_executable(iiJson_ontology tests/ontology.cpp)
    target_link_libraries(iiJson_ontology PRIVATE iiJson::Ontology)
    add_test(NAME iiJson.Ontology COMMAND iiJson_ontology)
    add_executable(iiJson_domain_example example/domain-schema.cpp)
    target_link_libraries(iiJson_domain_example PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.DomainExample COMMAND iiJson_domain_example)
    add_executable(iiJson_legacy_duplicate_example example/legacy-duplicate-policy.cpp)
    target_link_libraries(iiJson_legacy_duplicate_example PRIVATE iiJson::iiJson)
    add_test(NAME iiJson.LegacyDuplicateExample COMMAND iiJson_legacy_duplicate_example)
    add_executable(iiJson_ontology_example example/creation-ontology.cpp)
    target_link_libraries(iiJson_ontology_example PRIVATE iiJson::Ontology)
    add_test(NAME iiJson.OntologyExample COMMAND iiJson_ontology_example)
    add_test(NAME iiJson.Boundary COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py" "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
install(TARGETS iiJson iiJsonOntology EXPORT iiJsonTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES
    src/iiJson/Json.h
    src/iiJson/Value.h
    src/iiJson/Schema.h
    src/iiJson/Ontology.h
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiJson)
install(EXPORT iiJsonTargets NAMESPACE iiJson:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiJson)
configure_package_config_file(cmake/iiJsonConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiJsonConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiJson)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiJsonConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiJsonConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiJsonConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiJson)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/iiJson)

install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiJson/docs
    PATTERN "*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
install(DIRECTORY example/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiJson/example
    PATTERN "*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
